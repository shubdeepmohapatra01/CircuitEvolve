\begin{frame}{Never trained: 6 of 9 circuits beat the planner}

\includegraphics[width=0.94\textwidth]{circuitevolve_figs/bar_recon_frozen.png}

\begin{itemize}
  \setlength\itemsep{3pt}
  \item Playbook frozen, reflector off: gf2$^4$\_mult $-44\%$, mod\_red\_21 $-21\%$, adder\_8 $-11\%$
\end{itemize}
\vspace{-0.1cm}
\raggedright\scriptsize
Labels: T saved vs the planner. Planner and LLM: relaxed; TKET, pyzx: exact (not computed for ham15-med, gf2$^6$\_mult, hwb6, mod\_red\_21).

\end{frame}


\begin{frame}{What the LLM found}

\begin{itemize}
  \setlength\itemsep{10pt}
  \item \textbf{Re-identification:} the 12 Toffolis of barenco\_tof\_5 are one C$^5$X; rebuilt as a 4-AND ladder, $4k-4 = 16$ T
  \item \textbf{AND-preserving pairing:} rc\_adder\_6 Toffolis whose control is rewritten in between, as long as $c_0 c_1$ is preserved
  \item \textbf{Graduated:} the pairing is now exact code (G2) and reproduces 24 T and 44 T with zero LLM calls
\end{itemize}

\vspace{0.3cm}
\centering
\alert{LLM discovers, verifier certifies, rule becomes deterministic}

\end{frame}

\section{Angle}
% =====================================================================

\begin{frame}{Beyond the G1 pass, the LLM adds almost nothing}

\centering
\includegraphics[width=0.94\textwidth]{circuitevolve_figs/bar_angle.png}

\begin{itemize}
  \setlength\itemsep{3pt}
  \item G1 (same-Pauli merge, no LLM): $-10$ to $-15\%$ on qft, hhl, qpe, qae
  \item LLM on top: $-2$ T on qft\_8, $-147$ T on hhl\_8, nothing elsewhere
\end{itemize}
\vspace{-0.1cm}
\raggedright\scriptsize
$\varepsilon = 10^{-3}$. BQSKit-FT baseline without DPF (B2) equals TKET on all 8 circuits.

\end{frame}


\begin{frame}{Why windows had no headroom}

\begin{itemize}
  \setlength\itemsep{10pt}
  \item \textbf{TKET already saturated the windows.} heisenberg\_6, hubbard\_8, qml\_13: zero accepts in every arm
  \item \textbf{Rotations are expensive, merges are rare.} $\approx 3\log_2(1/\varepsilon_{rz})$ T per synthesized $R_z$ (about 46 T each on qft\_8); snapping an angle spends error budget
  \item \textbf{The remaining gains are global.} pyzx and DPF act on the whole circuit; a $\leq 4$-qubit window cannot see them
\end{itemize}

\vspace{0.3cm}
\centering
\alert{Local windows cannot reach global gains}

\end{frame}


\begin{frame}{Reconstruction learned rules; angle did not}

\centering
\includegraphics[width=0.74\textwidth]{circuitevolve_figs/bar_rules.png}

\begin{itemize}
  \setlength\itemsep{3pt}
  \item Reconstruction: 15 rules promoted by the A/B test; the frozen playbook transferred to unseen circuits
  \item Angle: 0 promoted; arms A (playbook), B (placebo), C (none) end at the same T on all four DEV circuits run in every arm
\end{itemize}

\end{frame}

\section{Combining the stages}
% =====================================================================

\begin{frame}{Stages compose: reconstruction first, then angle}

\centering
\includegraphics[width=0.94\textwidth]{circuitevolve_figs/bar_cascade.png}

\begin{itemize}
  \setlength\itemsep{3pt}
  \item Mixed circuit: angle alone $162$ T, cascade $130$ T (angle metric)
  \item Toffoli segment $56 \to 16$ named T; the angle stage then finds nothing more
\end{itemize}
\vspace{-0.1cm}
\raggedright\scriptsize
Zero-LLM demo (deterministic planner + G1). An LLM cascade run is still to do.

\end{frame}


\begin{frame}{Conclusions}

\begin{itemize}
  \setlength\itemsep{10pt}
  \item Reconstruction: the LLM beats the deterministic planner on \textbf{8 of 14} circuits, \textbf{6 of 9} never trained on
  \item Rules learned there transfer, and one discovery became \textbf{exact code} (G2)
  \item Angle windows: no headroom after TKET + G1; the LLM adds $\leq 0.5\%$
  \item Next: let the LLM propose \textbf{passes}, not windows --- G1 started as a seed rule
\end{itemize}

\end{frame}
